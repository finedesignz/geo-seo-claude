AuditStatus = 'queued' | 'running' | 'done' | 'failed'
AuditJob = row shape of the audits table (all D-06 columns, typed; score number|null, findings FindingsShape|null, the timestamptz columns as Date|null)
FindingsShape = structured findings persisted by the worker; reuse @geo/core result types
  (e.g. RobotsResult/CitabilityResult/SchemaTemplateResult/RenderingResult from packages/core/src/types.ts)
  composed into one findings object — import from @geo/core, do not re-declare.

DAL functions (D-10):
  insertJob(input: { url; normalizedUrl; urlHash; callbackUrl? }) -> AuditJob   // status defaults 'queued'
  claimNextJob(leaseSecs = 300) -> AuditJob | null                              // SKIP LOCKED txn
  completeJob(id, score: number, findings: FindingsShape) -> void              // running→done, sets finished_at
  failJob(id, errorCode: string) -> void                                       // running→failed, sets finished_at
  getJob(id) -> AuditJob | null
  listJobs(pagination: { limit; offset }) -> AuditJob[]                         // ORDER BY created_at DESC
  findRecentByUrlHash(urlHash, ttlMs) -> AuditJob | null                        // dedup, most recent within TTL
  reclaimExpired(maxAttempts = 3) -> number                                    // count of rows reclaimed/failed

@geo/core type exports available (packages/core/src/types.ts):
  RobotsResult, LlmsTxtResult, SchemaTemplateResult, StructuredDataValidationResult,
  CitabilityResult, RenderingResult
